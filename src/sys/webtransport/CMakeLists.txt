# Pinned, statically linked C dependencies; no runtime package downloads.
cmake_minimum_required(VERSION 3.18)
project(PDGWebTransport LANGUAGES C CXX)
set(CMAKE_POLICY_VERSION_MINIMUM 3.10)
include(FetchContent)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(ENABLE_TESTING OFF CACHE BOOL "" FORCE)
set(ENABLE_PROGRAMS OFF CACHE BOOL "" FORCE)
# Picoquic declares install exports that reference these targets. Keep their
# export declarations available even though PDG never runs an install step.
set(DISABLE_PACKAGE_CONFIG_AND_INSTALL OFF CACHE BOOL "" FORCE)
FetchContent_Declare(mbedtls GIT_REPOSITORY https://github.com/Mbed-TLS/mbedtls.git
    GIT_TAG 068ff080b369adfac81509f9b57b2afabaf82dc5 GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(mbedtls)
set(MBEDTLS_INCLUDE_DIR "${mbedtls_SOURCE_DIR}/include" CACHE PATH "" FORCE)
set(MBEDTLS_LIBRARY mbedtls CACHE STRING "" FORCE)
set(MBEDTLS_CRYPTO mbedcrypto CACHE STRING "" FORCE)
set(MBEDTLS_X509 mbedx509 CACHE STRING "" FORCE)
set(MBEDTLS_INCLUDE_DIRS "${mbedtls_SOURCE_DIR}/include")
set(MBEDTLS_LIBRARIES mbedtls mbedx509 mbedcrypto)
set(MBEDTLS_FOUND TRUE)
set(WITH_OPENSSL OFF CACHE BOOL "" FORCE)
set(WITH_MBEDTLS ON CACHE BOOL "" FORCE)
set(PICOQUIC_FETCH_PTLS ON CACHE BOOL "" FORCE)
set(PICOQUIC_FETCH_PTLS_TAG bfa67875982afc4c24f21e146cef4747fa189c2f CACHE STRING "" FORCE)
# Build only the TLS libraries: upstream's CMake also configures host tools,
# USDT and optional system compression/TLS libraries unsuitable for iOS.
FetchContent_Declare(picotls GIT_REPOSITORY https://github.com/h2o/picotls.git
    GIT_TAG bfa67875982afc4c24f21e146cef4747fa189c2f SOURCE_SUBDIR pdg-no-cmake)
FetchContent_MakeAvailable(picotls)
set(PTLS_INCLUDE_DIR "${picotls_SOURCE_DIR}/include" CACHE PATH "" FORCE)
add_library(picotls-core STATIC ${picotls_SOURCE_DIR}/lib/hpke.c ${picotls_SOURCE_DIR}/lib/picotls.c ${picotls_SOURCE_DIR}/lib/pembase64.c)
set(pdg_cifra aes blockwise chacha20 chash curve25519 drbg hmac gcm gf128 modes poly1305 sha256 sha512)
set(pdg_minicrypto_sources ${picotls_SOURCE_DIR}/deps/micro-ecc/uECC.c)
foreach(name IN LISTS pdg_cifra)
    list(APPEND pdg_minicrypto_sources ${picotls_SOURCE_DIR}/deps/cifra/src/${name}.c)
endforeach()
foreach(name cifra cifra/x25519 cifra/chacha20 cifra/aes128 cifra/aes256 cifra/random minicrypto-pem uecc asn1 ffx)
    list(APPEND pdg_minicrypto_sources ${picotls_SOURCE_DIR}/lib/${name}.c)
endforeach()
add_library(picotls-minicrypto STATIC ${pdg_minicrypto_sources})
target_link_libraries(picotls-minicrypto PUBLIC picotls-core)
foreach(target picotls-core picotls-minicrypto)
    target_include_directories(${target} PUBLIC "$<BUILD_INTERFACE:${picotls_SOURCE_DIR}/include>" PRIVATE
        ${picotls_SOURCE_DIR}/deps/cifra/src/ext ${picotls_SOURCE_DIR}/deps/cifra/src
        ${picotls_SOURCE_DIR}/deps/micro-ecc)
    if(WIN32)
        target_include_directories(${target} PUBLIC
            "$<BUILD_INTERFACE:${picotls_SOURCE_DIR}/picotlsvs/picotls>")
    endif()
endforeach()
set(picoquic_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(BUILD_DEMO OFF CACHE BOOL "" FORCE)
set(BUILD_PQBENCH OFF CACHE BOOL "" FORCE)
set(BUILD_LOGREADER OFF CACHE BOOL "" FORCE)
set(BUILD_PICO_SIM OFF CACHE BOOL "" FORCE)
set(DISABLE_DEBUG_PRINTF ON CACHE BOOL "" FORCE)
FetchContent_Declare(picoquic GIT_REPOSITORY https://github.com/private-octopus/picoquic.git
    GIT_TAG 0a847bd58f5dd4f19238876a52f0b8600f066b18 GIT_SHALLOW FALSE)
FetchContent_MakeAvailable(picoquic)
# Disabling upstream debug output leaves diagnostic locals unused in sockloop.c.
# Keep the exception at that dependency source rather than hiding PDG warnings.
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    set_source_files_properties("${picoquic_SOURCE_DIR}/picoquic/sockloop.c"
        TARGET_DIRECTORY picoquic-core PROPERTIES
        COMPILE_OPTIONS "-Wno-unused-variable;-Wno-unused-but-set-variable")
endif()
# The pinned verifier frees the parsed chain's contents but leaks its heap root.
# Compile a corrected build copy; leave the downloaded dependency unchanged.
set(pdg_sign_source "${picoquic_SOURCE_DIR}/picoquic_mbedtls/ptls_mbedtls_sign.c")
file(READ "${pdg_sign_source}" pdg_sign_text)
set(pdg_leak_pattern "mbedtls_x509_crt_free(chain_head);\n    }\n    return ret;")
string(FIND "${pdg_sign_text}" "${pdg_leak_pattern}" pdg_leak_offset)
if(pdg_leak_offset EQUAL -1)
    message(FATAL_ERROR "Review the certificate verifier patch after changing picoquic's pin")
endif()
string(REPLACE "${pdg_leak_pattern}" "mbedtls_x509_crt_free(chain_head);\n        free(chain_head);\n    }\n    return ret;" pdg_sign_text "${pdg_sign_text}")
file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/ptls_mbedtls_sign.c" "${pdg_sign_text}")
get_target_property(pdg_quic_sources picoquic-core SOURCES)
list(REMOVE_ITEM pdg_quic_sources "${pdg_sign_source}" "picoquic_mbedtls/ptls_mbedtls_sign.c")
list(APPEND pdg_quic_sources "${CMAKE_CURRENT_BINARY_DIR}/ptls_mbedtls_sign.c")
set_property(TARGET picoquic-core PROPERTY SOURCES "${pdg_quic_sources}")
set(ENABLE_CJSON_TEST OFF CACHE BOOL "" FORCE)
set(ENABLE_CJSON_UTILS OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
FetchContent_Declare(cjson GIT_REPOSITORY https://github.com/DaveGamble/cJSON.git
    GIT_TAG c859b25da02955fef659d658b8f324b5cde87be3 GIT_SHALLOW FALSE
    SOURCE_SUBDIR pdg-no-cmake)
FetchContent_MakeAvailable(cjson)
# Configure a build-local copy with an explicit policy range. This fixes the
# pin's old CMake compatibility warning, including on CMake 3.x, without editing
# downloaded sources or source overrides used by another build.
set(pdg_cjson_source "${CMAKE_CURRENT_BINARY_DIR}/cjson-source")
file(READ "${cjson_SOURCE_DIR}/CMakeLists.txt" pdg_cjson_cmake)
set(pdg_cjson_minimum "cmake_minimum_required(VERSION 3.0)")
string(FIND "${pdg_cjson_cmake}" "${pdg_cjson_minimum}" pdg_cjson_minimum_offset)
if(pdg_cjson_minimum_offset EQUAL -1)
    message(FATAL_ERROR "Review the cJSON CMake policy patch after changing its pin")
endif()
string(REPLACE "${pdg_cjson_minimum}" "cmake_minimum_required(VERSION 3.0...3.10)"
    pdg_cjson_cmake "${pdg_cjson_cmake}")
file(COPY "${cjson_SOURCE_DIR}/" DESTINATION "${pdg_cjson_source}" PATTERN ".git" EXCLUDE)
file(WRITE "${pdg_cjson_source}/CMakeLists.txt" "${pdg_cjson_cmake}")
add_subdirectory("${pdg_cjson_source}" "${cjson_BINARY_DIR}")
add_library(pdg_webtransport STATIC pdg_webtransport.cpp)
target_compile_features(pdg_webtransport PUBLIC cxx_std_17)
if(MSVC)
    # The pinned Picotls header uses C99 compound literals that MSVC rejects in C++.
    # Keep its C targets on the original header and adapt only the C++ bridge copy.
    file(READ "${picotls_SOURCE_DIR}/include/picotls.h" pdg_picotls_header)
    foreach(type ptls_log_getsni_t ptls_buffer_t ptls_hash_context_t)
        string(FIND "${pdg_picotls_header}" "(${type}){" found)
        if(found EQUAL -1)
            message(FATAL_ERROR "Review the Picotls C++ header adaptation after changing its pin: ${type}")
        endif()
        string(REPLACE "(${type}){" "${type}{" pdg_picotls_header "${pdg_picotls_header}")
    endforeach()
    set(pdg_picotls_cxx_dir "${CMAKE_CURRENT_BINARY_DIR}/picotls-cxx-compat")
    file(MAKE_DIRECTORY "${pdg_picotls_cxx_dir}")
    file(WRITE "${pdg_picotls_cxx_dir}/picotls.h" "${pdg_picotls_header}")
    target_include_directories(pdg_webtransport BEFORE PRIVATE "${pdg_picotls_cxx_dir}")
endif()
if(CMAKE_SYSTEM_NAME STREQUAL "iOS")
    target_compile_definitions(pdg_webtransport PRIVATE PDG_WEBTRANSPORT_CLIENT_ONLY)
endif()
if(WIN32)
    target_sources(picotls-core PRIVATE ${picotls_SOURCE_DIR}/picotlsvs/picotls/wintimeofday.c)
    target_compile_definitions(picotls-core PRIVATE _WINDOWS _CRT_SECURE_NO_WARNINGS)
    target_compile_definitions(picotls-minicrypto PRIVATE _WINDOWS _CRT_SECURE_NO_WARNINGS)
endif()
find_package(Threads REQUIRED)
target_link_libraries(pdg_webtransport PUBLIC picohttp-core picoquic-core cjson Threads::Threads)
target_include_directories(pdg_webtransport PUBLIC ${CMAKE_CURRENT_SOURCE_DIR} PRIVATE ${cjson_SOURCE_DIR})
if(APPLE)
    target_link_libraries(pdg_webtransport PUBLIC "-framework Security" "-framework CoreFoundation")
elseif(WIN32)
    target_compile_definitions(pdg_webtransport PRIVATE NOMINMAX)
    target_link_libraries(pdg_webtransport PUBLIC crypt32 ws2_32)
endif()
option(PDG_WEBTRANSPORT_TEST_DRIVER "Build the native integration test driver" OFF)
if(PDG_WEBTRANSPORT_TEST_DRIVER)
    add_executable(pdg_webtransport_test_driver ../../../test/cxx/webtransport_bridge_host.cpp)
    target_link_libraries(pdg_webtransport_test_driver PRIVATE pdg_webtransport)
endif()
# One archive keeps Xcode and node-gyp consumers independent of dependency layout.
set(pdg_bundle_targets pdg_webtransport picohttp-core picoquic-log picoquic-core
    picotls-minicrypto picotls-core mbedtls mbedx509 mbedcrypto everest p256m cjson)
set(pdg_bundle_libraries "")
foreach(target IN LISTS pdg_bundle_targets)
    string(APPEND pdg_bundle_libraries "\"$<TARGET_FILE:${target}>\" ")
endforeach()
set(pdg_bundle_platform "ar")
if(APPLE)
    set(pdg_bundle_platform "Apple")
elseif(MSVC)
    set(pdg_bundle_platform "MSVC")
endif()
set(PDG_WEBTRANSPORT_ARCHIVE "${CMAKE_CURRENT_BINARY_DIR}/${CMAKE_STATIC_LIBRARY_PREFIX}pdg_webtransport_all${CMAKE_STATIC_LIBRARY_SUFFIX}")
file(GENERATE OUTPUT "${CMAKE_CURRENT_BINARY_DIR}/bundle-$<CONFIG>.cmake" CONTENT
    "set(PDG_BUNDLE_ARCHIVES ${pdg_bundle_libraries})\nset(PDG_BUNDLE_AR \"${CMAKE_AR}\")\nset(PDG_BUNDLE_OUTPUT \"${PDG_WEBTRANSPORT_ARCHIVE}\")\nset(PDG_BUNDLE_PLATFORM \"${pdg_bundle_platform}\")\ninclude(\"${CMAKE_CURRENT_SOURCE_DIR}/bundle.cmake\")\n")
add_custom_command(OUTPUT "${PDG_WEBTRANSPORT_ARCHIVE}"
    COMMAND "${CMAKE_COMMAND}" -P "${CMAKE_CURRENT_BINARY_DIR}/bundle-$<CONFIG>.cmake"
    DEPENDS ${pdg_bundle_targets} "${CMAKE_CURRENT_SOURCE_DIR}/bundle.cmake")
add_custom_target(pdg_webtransport_bundle ALL DEPENDS "${PDG_WEBTRANSPORT_ARCHIVE}")
