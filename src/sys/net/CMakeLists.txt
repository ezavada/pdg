cmake_minimum_required(VERSION 3.18)
project(PDGNativeNetwork LANGUAGES C CXX)
if(EMSCRIPTEN)
    message(FATAL_ERROR "pdg-net is for native C++ applications only")
endif()
# Match the native PDG core's static MSVC runtime, including dependencies.
if(MSVC AND NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
endif()
add_subdirectory(../webtransport webtransport)
FetchContent_GetProperties(mbedtls)
FetchContent_GetProperties(cjson)
add_library(pdg_net STATIC network.cpp socket.cpp webtransport.cpp)
target_compile_features(pdg_net PUBLIC cxx_std_20)
target_include_directories(pdg_net PUBLIC ../../inc PRIVATE .. ${mbedtls_SOURCE_DIR}/include ${cjson_SOURCE_DIR})
target_link_libraries(pdg_net PUBLIC pdg_webtransport PRIVATE mbedtls mbedx509 mbedcrypto cjson)
if(WIN32)
    target_compile_definitions(pdg_net PRIVATE NOMINMAX)
    target_link_libraries(pdg_net PUBLIC ws2_32 crypt32)
endif()
